Combine `ser-round`, `ser-rounder`, or `ser-roundest` with buttons and inputs.

<BorderRadiusGallery />

```html
<!-- Round buttons -->
<button class="ser-btn-primary-small ser-round">Round</button>
<button class="ser-btn-primary-small ser-rounder">Rounder</button>
<button class="ser-btn-primary-small ser-roundest">Roundest</button>

<!-- Round inputs -->
<input class="ser-input-primary-small-outlined ser-round" />
<input class="ser-input-primary-medium-outlined ser-rounder" />
<input class="ser-input-secondary-medium-outlined ser-roundest" />
```
